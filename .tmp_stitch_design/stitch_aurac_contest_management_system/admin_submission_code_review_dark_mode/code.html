#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

// Core optimization logic for node traversal
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, m;
    if (!(cin >> n >> m)) return 0;

    vector<long long> weights(n);
    for (int i = 0; i < n; ++i) {
        cin >> weights[i];
    }

    sort(weights.begin(), weights.end());

    long long optimal_path = 0;
    for (int i = 0; i < m; ++i) {
        optimal_path += weights[n - 1 - i];
    }

    cout << optimal_path << "\n";
    return 0;
}
